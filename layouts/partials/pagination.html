{{- /* pagination — previous / next and numbered pages for a paginated list (the blog and each tag page). Adapted from ofreport's partial, in this site's colours. Renders nothing until there is a second page. Numbers show the current page ± 2, plus the first and last with an ellipsis across any gap; on phones only Previous and Next show. Takes the page context, after the template has called .Paginate. */ -}} {{- $paginator := .Paginator -}} {{- if gt $paginator.TotalPages 1 -}} {{- $current := $paginator.PageNumber -}} {{- $total := $paginator.TotalPages -}} {{- $start := math.Max 1 (sub $current 2) -}} {{- $end := math.Min $total (add $current 2) -}} {{- $edge := "rounded-eto px-1 pt-4 text-sm font-semibold text-warm-600 transition-colors hover:text-eto-600 focus-ring" -}} {{- $number := "inline-flex items-center border-t-2 border-transparent px-4 pt-4 text-sm font-semibold text-warm-600 transition-colors hover:border-warm-300 hover:text-eto-600 focus-ring" -}} {{- $gap := "inline-flex items-center px-4 pt-4 text-sm text-warm-600" -}}
{{- with $paginator.Prev }} ← Previous {{- end }}
{{- range $paginator.Pagers }} {{- $n := .PageNumber }} {{- if and (eq $n 1) (gt $start 1) }} 1 {{- if gt $start 2 }}…{{ end }} {{- end }} {{- if and (ge $n (int $start)) (le $n (int $end)) }} {{- if eq $n $current }} {{ $n }} {{- else }} {{ $n }} {{- end }} {{- end }} {{- if and (eq $n $total) (lt $end $total) }} {{- if lt $end (sub $total 1) }}…{{ end }} {{ $total }} {{- end }} {{- end }}
{{- with $paginator.Next }} Next → {{- end }}
{{- end -}}
